```swift
import Foundation

extension JSONDecoder {
    /// Décodeur des réponses lolesports : dates ISO 8601 avec ou sans fractions de seconde.
    public static func lolesports() -> JSONDecoder {
        let decoder = JSONDecoder()
        decoder.dateDecodingStrategy = .custom { decoder in
            let container = try decoder.singleValueContainer()
            let string = try container.decode(String.self)
            if let date = LoLDate.parse(string) { return date }
            throw DecodingError.dataCorruptedError(in: container, debugDescription: "Date ISO 8601 invalide : \(string)")
        }
        return decoder
    }
}

/// Lecture et écriture des dates ISO 8601 de l'API.
public enum LoLDate {
    public static func parse(_ string: String) -> Date? {
        let withFraction = ISO8601DateFormatter()
        withFraction.formatOptions = [.withInternetDateTime, .withFractionalSeconds]
        if let date = withFraction.date(from: string) { return date }
        let plain = ISO8601DateFormatter()
        plain.formatOptions = [.withInternetDateTime]
        return plain.date(from: string)
    }

    /// Format attendu par `startingTime` du flux livestats : `2026-10-04T20:38:20.000Z`.
    public static func format(_ date: Date) -> String {
        let formatter = ISO8601DateFormatter()
        formatter.formatOptions = [.withInternetDateTime, .withFractionalSeconds]
        return formatter.string(from: date)
    }
}

/// Tableau tolérant : un élément illisible est ignoré au lieu de faire échouer tout le décodage.
struct LossyArray<Element: Decodable>: Decodable {
    var elements: [Element]

    init(from decoder: any Decoder) throws {
        var container = try decoder.unkeyedContainer()
        var elements: [Element] = []
        while !container.isAtEnd {
            if let element = try? container.decode(Element.self) {
                elements.append(element)
            } else {
                _ = try? container.decode(Skip.self)
            }
        }
        self.elements = elements
    }

    /// Consomme un élément sans le lire, pour avancer dans le tableau.
    private struct Skip: Decodable {
        init(from decoder: any Decoder) throws {}
    }
}
```

public struct TeamStats: Codable, Sendable, Equatable, Hashable {
    public var kills: Int
    public var gold: Int
    public var towers: Int
    public var inhibitors: Int
    public var barons: Int

    public var dragons: [String]

    public init(kills: Int = 0, gold: Int = 0, towers: Int = 0, inhibitors: Int = 0, barons: Int = 0, dragons: [String] = []) {
        self.kills = kills
        self.gold = gold
        self.towers = towers
        self.inhibitors = inhibitors
        self.barons = barons
        self.dragons = dragons
    }

    init(_ team: FrameTeam) {
        self.init(kills: team.totalKills, gold: team.totalGold, towers: team.towers,
                  inhibitors: team.inhibitors, barons: team.barons, dragons: team.dragons)
    }
}

/// Une ligne joueur de l'écran Détail.
public struct PlayerLine: Sendable, Equatable {
    public let role: String
    public let champion: String
    public let name: String
    public let level: Int
    public let kills: Int
    public let deaths: Int
    public let assists: Int
    public let creepScore: Int
    public let gold: Int

    static func lines(_ participants: [FrameParticipant], metadata: [ParticipantMetadata]) -> [PlayerLine] {
        participants.map { participant in
            let meta = metadata.first { $0.participantId == participant.participantId }
            return PlayerLine(
                role: meta?.role ?? "",
                champion: meta?.championId ?? "?",
                name: meta?.summonerName ?? "",
                level: participant.level,
                kills: participant.kills,
                deaths: participant.deaths,
                assists: participant.assists,
                creepScore: participant.creepScore,
                gold: participant.totalGold
            )
        }
    }
}

public enum GamePhase: String, Sendable, Equatable {
    case inGame, paused, finished

    init(apiValue: String) {
        switch apiValue {
        case "finished": self = .finished
        case "paused": self = .paused
        default: self = .inGame
        }
    }
}

/// État d'une game à la dernière frame reçue, côté bleu et côté rouge.
public struct GameSnapshot: Sendable, Equatable {
    public let gameId: String
    public let phase: GamePhase
    public let timestamp: Date
    public let blueTeamId: String
    public let redTeamId: String
    public let blue: TeamStats
    public let red: TeamStats
    public let bluePlayers: [PlayerLine]
    public let redPlayers: [PlayerLine]

    /// Stats de l'équipe `teamId`, quel que soit son côté.
    public func stats(for teamId: String) -> TeamStats? {
        if teamId == blueTeamId { return blue }
        if teamId == redTeamId { return red }
        return nil
    }

    public func players(for teamId: String) -> [PlayerLine]? {
        if teamId == blueTeamId { return bluePlayers }
        if teamId == redTeamId { return redPlayers }
        return nil
    }
}

extension LiveWindow {
    /// Dernière frame de la fenêtre, avec champions et rôles des joueurs.
    public func latestSnapshot() -> GameSnapshot? {
        guard let frame = frames.last else { return nil }
        return GameSnapshot(
            gameId: esportsGameId,
            phase: GamePhase(apiValue: frame.gameState),
            timestamp: frame.rfc460Timestamp,
            blueTeamId: gameMetadata.blueTeamMetadata.esportsTeamId,
            redTeamId: gameMetadata.redTeamMetadata.esportsTeamId,
            blue: TeamStats(frame.blueTeam),
            red: TeamStats(frame.redTeam),
            bluePlayers: PlayerLine.lines(frame.blueTeam.participants, metadata: gameMetadata.blueTeamMetadata.participantMetadata),
            redPlayers: PlayerLine.lines(frame.redTeam.participants, metadata: gameMetadata.redTeamMetadata.participantMetadata)
        )
    }

    /// Horodatage de la première frame : le début de la game si la fenêtre a été demandée sans `startingTime`.
    public var firstFrameTime: Date? { frames.first?.rfc460Timestamp }
}

    }

import Foundation

import FoundationNetworking

public struct HTTPResponse: Sendable {
    public let status: Int
    public let body: Data

    public init(status: Int, body: Data) {
        self.status = status
        self.body = body
    }
}

/// Accès réseau, remplaçable dans les tests.
public protocol HTTPTransport: Sendable {
    func get(_ url: URL, headers: [String: String]) async throws -> HTTPResponse
}

/// Transport réel. Utilise `dataTask` (disponible partout, y compris Linux).
public struct URLSessionTransport: HTTPTransport {
    public init() {}

    public func get(_ url: URL, headers: [String: String]) async throws -> HTTPResponse {
        var request = URLRequest(url: url)
        request.timeoutInterval = 15
        for (name, value) in headers {
            request.setValue(value, forHTTPHeaderField: name)
        }
        let finalRequest = request
        return try await withCheckedThrowingContinuation { continuation in
            URLSession.shared.dataTask(with: finalRequest) { data, response, error in
                if let error {
                    continuation.resume(throwing: error)
                    return
                }
                let status = (response as? HTTPURLResponse)?.statusCode ?? 0
                continuation.resume(returning: HTTPResponse(status: status, body: data ?? Data()))
            }.resume()

            throw LoLAPIError.noData
        case 404:

        let response = try await transport.get(components.url!, headers: ["x-api-key": Self.publicAPIKey])
        guard response.status == 200 else { throw LoLAPIError.http(status: response.status) }
        return try JSONDecoder.lolesports().decode(T.self, from: response.body)

/// Une équipe suivie et ses réglages.
public struct FollowedTeam: Codable, Sendable, Equatable, Identifiable {
    public static let defaultColorHex = "#6FD3FF"

    public let id: String
    public var code: String
    public var name: String
    public var image: String?
    public var homeLeagueName: String?
    public var spoilerFree: Bool
    /// Couleur tirée du logo (calculée par l'appli avec `TeamColor`).
    public var autoColorHex: String?
    /// Couleur choisie à la main ; prioritaire.
    public var colorOverrideHex: String?

    public init(team: Team) {
        id = team.id
        code = team.code
        name = team.name
        image = team.image
        homeLeagueName = team.homeLeague?.name
        spoilerFree = false
        autoColorHex = nil
        colorOverrideHex = nil
    }

    /// Couleur à utiliser partout (Vitrail : une teinte = une équipe).
    public var colorHex: String { colorOverrideHex ?? autoColorHex ?? Self.defaultColorHex }
}

extension MatchTeam {
    /// Vrai si cette équipe d'un match est l'équipe suivie : id, sinon nom, sinon logo. Jamais le code seul (spec §4.2).
    public func isSame(as team: FollowedTeam) -> Bool {
        if let id { return id == team.id }
        if name.lowercased() == team.name.lowercased() { return true }
        if let image, let other = team.image { return image == other }
        return false
    }
}

/// Les équipes suivies, dans l'ordre où elles ont été suivies.
public struct FollowList: Codable, Sendable, Equatable {
    public private(set) var teams: [FollowedTeam]

    public init(teams: [FollowedTeam] = []) {
        self.teams = teams
    }

    public func contains(id: String) -> Bool {
        teams.contains { $0.id == id }
    }

    public func team(id: String) -> FollowedTeam? {
        teams.first { $0.id == id }
    }

    /// Équipes suivies présentes dans ce match, dans l'ordre de suivi.
    public func followed(in matchTeams: [MatchTeam]) -> [FollowedTeam] {
        teams.filter { followed in matchTeams.contains { $0.isSame(as: followed) } }
    }

    public func involves(_ matchTeams: [MatchTeam]) -> Bool {
        !followed(in: matchTeams).isEmpty
    }

    public mutating func follow(_ team: Team) {
        guard !contains(id: team.id) else { return }
        teams.append(FollowedTeam(team: team))
    }

    public mutating func unfollow(id: String) {
        teams.removeAll { $0.id == id }
    }

    public mutating func setSpoilerFree(_ on: Bool, id: String) {
        update(id) { $0.spoilerFree = on }
    }

    public mutating func setAutoColor(_ hex: String, id: String) {
        update(id) { $0.autoColorHex = hex }
    }

    public mutating func setColorOverride(_ hex: String?, id: String) {
        update(id) { $0.colorOverrideHex = hex }
    }

    /// Met à jour nom, code, logo et ligue depuis `getTeams` (renommages de sponsor).
    public mutating func refresh(from catalog: [Team]) {
        for team in catalog {
            update(team.id) {
                $0.code = team.code
                $0.name = team.name
                $0.image = team.image
                $0.homeLeagueName = team.homeLeague?.name
            }
        }
    }

    private mutating func update(_ id: String, _ change: (inout FollowedTeam) -> Void) {
        guard let index = teams.firstIndex(where: { $0.id == id }) else { return }
        change(&teams[index])
    }
}

/// Une carte Vitrail de l'onglet Matchs (spec §5.1) : une par équipe suivie.
public struct TeamCard: Sendable, Equatable, Identifiable {
    public let team: FollowedTeam
    public let live: MatchEvent?
    public let next: MatchEvent?
    public let last: MatchEvent?

    public var id: String { team.id }
}

public enum TeamCards {
    /// Cartes triées : équipes en direct d'abord (ordre de suivi), puis par date du prochain match, puis ordre de suivi.
    public static func build(follows: FollowList, schedule: [MatchEvent], live: [MatchEvent]) -> [TeamCard] {
        let liveIds = Set(live.map(\.id))
        let cards = follows.teams.map { team -> TeamCard in
            let involves: (MatchEvent) -> Bool = { event in event.match.teams.contains { $0.isSame(as: team) } }
            let liveMatch = live.first(where: involves)
                ?? schedule.first { involves($0) && $0.state == .inProgress }
            let next = schedule
                .filter { involves($0) && $0.state == .unstarted && !liveIds.contains($0.id) }
                .min { $0.startTime < $1.startTime }
            let last = schedule
                .filter { involves($0) && $0.state == .completed }
                .max { $0.startTime < $1.startTime }
            return TeamCard(team: team, live: liveMatch, next: next, last: last)
        }
        return cards.enumerated().sorted { a, b in
            switch (a.element.live != nil, b.element.live != nil) {
            case (true, false): return true
            case (false, true): return false
            case (true, true): return a.offset < b.offset
            case (false, false):
                switch (a.element.next?.startTime, b.element.next?.startTime) {
                case let (x?, y?): return x != y ? x < y : a.offset < b.offset
                case (.some, nil): return true
                case (nil, .some): return false
                case (nil, nil): return a.offset < b.offset
                }
            }
        }.map(\.element)
    }
}

import Foundation

/// La règle anti-spoiler, unique pour toutes les surfaces (spec §8).
public enum SpoilerPolicy {
    /// Masqué si au moins une équipe suivie du match est en anti-spoiler, sauf si le match a été révélé.
    public static func isMasked(_ teams: [MatchTeam], matchId: String, follows: FollowList, revealed: Set<String>) -> Bool {
        guard !revealed.contains(matchId) else { return false }
        return follows.followed(in: teams).contains { $0.spoilerFree }
    }
}

/// Ce que l'appli et les widgets ont le droit d'afficher pour un match.
public struct ScoreLine: Sendable, Equatable {
    public enum Status: Sendable, Equatable {
        case upcoming(Date)
        case live(gameNumber: Int?)
        case finished(winnerCode: String?)
    }

    public let matchId: String
    public let leagueName: String
    public let bestOf: Int
    public let teamCodes: [String]
    /// Victoires de chaque équipe ; `nil` si masqué ou pas commencé.
    public let score: [Int]?
    public let status: Status
    public let isMasked: Bool
}

public enum Presenter {
    public static func scoreLine(for event: MatchEvent, follows: FollowList, revealed: Set<String>) -> ScoreLine {
        let teams = event.match.teams
        let masked = SpoilerPolicy.isMasked(teams, matchId: event.id, follows: follows, revealed: revealed)
        let status: ScoreLine.Status
        let started: Bool
        switch event.state {
        case .inProgress:
            status = .live(gameNumber: masked ? nil : currentGameNumber(event))
            started = true
        case .completed:
            status = .finished(winnerCode: masked ? nil : teams.max { $0.gameWins < $1.gameWins }?.code)
            started = true
        case .unstarted, .unknown:
            status = .upcoming(event.startTime)
            started = false
        }
        return ScoreLine(
            matchId: event.id,
            leagueName: event.league.name,
            bestOf: event.bestOf,
            teamCodes: teams.map(\.code),
            score: (started && !masked) ? teams.map(\.gameWins) : nil,
            status: status,
            isMasked: masked
        )
    }

    /// Game en cours : celle marquée `inProgress`, sinon la suivante d'après les victoires.
    public static func currentGameNumber(_ event: MatchEvent) -> Int {
        if let game = event.match.games?.first(where: { $0.state == .inProgress }) { return game.number }
        return event.match.teams.reduce(0) { $0 + $1.gameWins } + 1
    }
}

import Foundation

/// Formats d'affichage en français, identiques sur toutes les plateformes.
public enum Format {
    /// 37442 → « 37,4k » ; 67995 → « 68k » ; 850 → « 850 ».
    public static func gold(_ value: Int) -> String {
        let magnitude = abs(value)
        guard magnitude >= 1000 else { return "\(value)" }
        let tenths = (magnitude + 50) / 100
        let sign = value < 0 ? "-" : ""
        let whole = tenths / 10
        let decimal = tenths % 10
        return decimal == 0 ? "\(sign)\(whole)k" : "\(sign)\(whole),\(decimal)k"
    }

    /// Écart signé : « +1,6k », « -800 », « 0 ».
    public static func goldDiff(_ diff: Int) -> String {
        diff > 0 ? "+" + gold(diff) : gold(diff)
    }

    private static let days = ["dim.", "lun.", "mar.", "mer.", "jeu.", "ven.", "sam."]
    private static let months = ["janv.", "févr.", "mars", "avr.", "mai", "juin", "juil.", "août", "sept.", "oct.", "nov.", "déc."]

    /// « jeu. 15 oct. · 20:00 » dans le fuseau donné.
    public static func matchDate(_ date: Date, timeZone: TimeZone = .current) -> String {
        var calendar = Calendar(identifier: .gregorian)
        calendar.timeZone = timeZone
        let parts = calendar.dateComponents([.weekday, .day, .month, .hour, .minute], from: date)
        let hour = parts.hour ?? 0
        let minute = parts.minute ?? 0
        let time = (hour < 10 ? "0" : "") + "\(hour):" + (minute < 10 ? "0" : "") + "\(minute)"
        return "\(days[(parts.weekday ?? 1) - 1]) \(parts.day ?? 0) \(months[(parts.month ?? 1) - 1]) · \(time)"
    }

    /// Compte à rebours court pour le widget rond : « 45 min », « 21h », « 3 j », « maintenant ».
    public static func countdown(from now: Date, to date: Date) -> String {
        let minutes = Int(date.timeIntervalSince(now) / 60)
        if minutes <= 0 { return "maintenant" }
        if minutes < 60 { return "\(minutes) min" }
        let hours = minutes / 60
        if hours < 48 { return "\(hours)h" }
        return "\(hours / 24) j"
    }
}

public struct LiveMatchState: Codable, Sendable, Equatable, Hashable {

        case upcoming, inGame, betweenGames, finished
    }

    /// Au-delà, la Live Activity affiche « màj il y a X min ».

    public var gameClockStart: Date?

    public var stats: [TeamStats]?
    public var statsAvailable: Bool
    public var spoilerFree: Bool
    public var winnerCode: String?
    public var updatedAt: Date

    public init(phase: Phase, seriesScore: [Int]?, gameNumber: Int?, gameClockStart: Date?, stats: [TeamStats]?,
                statsAvailable: Bool, spoilerFree: Bool, winnerCode: String?, updatedAt: Date) {
        self.phase = phase
        self.seriesScore = seriesScore
        self.gameNumber = gameNumber
        self.gameClockStart = gameClockStart
        self.stats = stats
        self.statsAvailable = statsAvailable
        self.spoilerFree = spoilerFree
        self.winnerCode = winnerCode
        self.updatedAt = updatedAt
    }

    public var staleDate: Date { updatedAt.addingTimeInterval(Self.staleAfter) }

    /// Version sans spoiler : seule la phase reste (« En direct », « Terminé »…).
    public func masked() -> LiveMatchState {
        var copy = self
        copy.seriesScore = nil
        copy.gameNumber = nil
        copy.gameClockStart = nil
        copy.stats = nil
        copy.winnerCode = nil
        copy.spoilerFree = true
        return copy
    }

import Foundation

/// Construit les données de la Live Activity à partir de l'API (spec §6).
public enum LiveMatchBuilder {
    public static func attributes(for event: MatchEvent, follows: FollowList) -> LiveMatchAttributes {
        LiveMatchAttributes(
            matchId: event.id,
            leagueName: event.league.name,
            bestOf: event.bestOf,
            teams: event.match.teams.map { team in
                let followed = follows.teams.first { team.isSame(as: $0) }
                return LiveMatchAttributes.Side(id: team.id ?? team.code, code: team.code, name: team.name, colorHex: followed?.colorHex)
            }
        )
    }

    /// - Parameters:
    ///   - snapshot: dernière frame du flux pour la game en cours (ignorée si elle concerne une autre game).
    ///   - firstFrameTime: première frame de la game en cours (fenêtre demandée sans `startingTime`).
    ///   - statsDisabled: `MatchEvent.statsDisabled` ou 404 sur le flux.
    public static func state(
        details: EventDetails,
        snapshot: GameSnapshot?,
        firstFrameTime: Date?,
        statsDisabled: Bool,
        spoilerFree: Bool,
        now: Date
    ) -> LiveMatchState {
        let teams = details.match.teams
        let wins = teams.map(\.gameWins)
        let played = wins.reduce(0, +)
        let needed = details.bestOf / 2 + 1
        let current = details.match.games.first { $0.state == .inProgress }
        let snap = snapshot.flatMap { $0.gameId == current?.id ? $0 : nil }
        let feedFinished = snap?.phase == .finished

        let phase: LiveMatchState.Phase
        if (wins.max() ?? 0) >= needed {
            phase = .finished
        } else if current != nil && !feedFinished {
            phase = .inGame
        } else if played > 0 || current != nil {
            phase = .betweenGames
        } else {
            phase = .upcoming
        }

        let gameNumber: Int?
        switch phase {
        case .upcoming: gameNumber = 1
        case .inGame: gameNumber = current?.number
        case .betweenGames: gameNumber = max(played + 1, (current?.number ?? 0) + (feedFinished ? 1 : 0))
        case .finished: gameNumber = nil
        }

        var stats: [TeamStats]?
        var clockStart: Date?
        if phase == .inGame, !statsDisabled, let snap {
            let perTeam = teams.compactMap { team in team.id.flatMap { snap.stats(for: $0) } }
            stats = perTeam.count == teams.count ? perTeam : nil
            if let firstFrameTime {
                clockStart = now.addingTimeInterval(-snap.timestamp.timeIntervalSince(firstFrameTime))
            }
        }

        let state = LiveMatchState(
            phase: phase,
            seriesScore: wins,
            gameNumber: gameNumber,
            gameClockStart: clockStart,
            stats: stats,
            statsAvailable: !statsDisabled,
            spoilerFree: false,
            winnerCode: phase == .finished ? teams.max { $0.gameWins < $1.gameWins }?.code : nil,
            updatedAt: now
        )
        return spoilerFree ? state.masked() : state
    }
}

/// Réglage « Live Activity » (spec §5.4).
public enum LiveActivityMode: String, Codable, Sendable {
    case automatic, manual
}

/// Décide quels matchs en direct ont une Live Activity (spec §6.4).
public enum LiveActivityPlanner {
    public static let maxConcurrent = 3

    /// Ids de matchs par priorité : épinglés (s'ils sont en direct), puis en automatique les matchs
    /// des équipes suivies, dans l'ordre de suivi, sauf ceux retirés à la main. Au plus `maxConcurrent`.
    public static func plan(
        live: [MatchEvent],
        follows: FollowList,
        mode: LiveActivityMode,
        pinned: [String],
        dismissed: Set<String>
    ) -> [String] {
        let liveIds = Set(live.map(\.id))
        var result = pinned.filter { liveIds.contains($0) }
        if mode == .automatic {
            let ranked = live.enumerated()
                .compactMap { position, event -> (rank: Int, position: Int, id: String)? in
                    let ranks = follows.teams.indices.filter { index in
                        event.match.teams.contains { $0.isSame(as: follows.teams[index]) }
                    }
                    return ranks.min().map { (rank: $0, position: position, id: event.id) }
                }
                .sorted { ($0.rank, $0.position) < ($1.rank, $1.position) }
            for candidate in ranked where !dismissed.contains(candidate.id) && !result.contains(candidate.id) {
                result.append(candidate.id)
            }
        }
        return Array(result.prefix(maxConcurrent))
    }
}

/// Couleur 8 bits par canal.
public struct RGB: Codable, Sendable, Equatable, Hashable {
    public var r: UInt8
    public var g: UInt8
    public var b: UInt8

    /// Accent de la DA (`#6FD3FF`), couleur par défaut d'une équipe.
    public static let accent = RGB(r: 0x6F, g: 0xD3, b: 0xFF)

    public init(r: UInt8, g: UInt8, b: UInt8) {
        self.r = r
        self.g = g
        self.b = b
    }

    public init?(hex: String) {
        let digits = hex.hasPrefix("#") ? String(hex.dropFirst()) : hex
        guard digits.count == 6, let value = UInt32(digits, radix: 16) else { return nil }
        self.init(r: UInt8((value >> 16) & 0xFF), g: UInt8((value >> 8) & 0xFF), b: UInt8(value & 0xFF))
    }

    public var hex: String {
        func pair(_ value: UInt8) -> String {
            let digits = String(value, radix: 16, uppercase: true)
            return digits.count == 1 ? "0" + digits : digits
        }
        return "#" + pair(r) + pair(g) + pair(b)
    }

    /// Luminosité perçue, de 0 (noir) à 1 (blanc).
    public var luminance: Double {
        (0.2126 * Double(r) + 0.7152 * Double(g) + 0.0722 * Double(b)) / 255
    }
}

/// Couleur d'une équipe à partir de son logo (spec §9).
public enum TeamColor {
    /// Luminosité minimale pour rester lisible sur le fond abysse `#0b1020`.
    public static let minLuminance = 0.5

    /// Couleur dominante saturée d'un logo. `rgba` : 4 octets par pixel.
    /// Ignore les pixels transparents, quasi gris, quasi blancs et quasi noirs.
    public static func dominant(rgba: [UInt8]) -> RGB? {
        var weights = [Double](repeating: 0, count: 36)
        var sums = [(r: Double, g: Double, b: Double, count: Double)](repeating: (0, 0, 0, 0), count: 36)
        var index = 0
        while index + 3 < rgba.count {
            let r = Double(rgba[index]), g = Double(rgba[index + 1]), b = Double(rgba[index + 2])
            let alpha = rgba[index + 3]
            index += 4
            guard alpha >= 128 else { continue }
            let maxChannel = max(r, g, b)
            let minChannel = min(r, g, b)
            let value = maxChannel / 255
            let saturation = maxChannel == 0 ? 0 : (maxChannel - minChannel) / maxChannel
            guard saturation >= 0.25, value >= 0.2 else { continue }
            let bin = min(35, Int(hue(r, g, b, maxChannel: maxChannel, minChannel: minChannel) / 10))
            weights[bin] += saturation * value
            sums[bin].r += r
            sums[bin].g += g
            sums[bin].b += b
            sums[bin].count += 1
        }
        guard let best = weights.indices.max(by: { weights[$0] < weights[$1] }), weights[best] > 0 else { return nil }
        let sum = sums[best]
        return RGB(r: UInt8((sum.r / sum.count).rounded()),
                   g: UInt8((sum.g / sum.count).rounded()),
                   b: UInt8((sum.b / sum.count).rounded()))
    }

    /// Éclaircit (mélange avec du blanc) jusqu'à atteindre `minLuminance`.
    public static func readable(_ color: RGB) -> RGB {
        var amount = 0.0
        var result = color
        while result.luminance < minLuminance && amount < 1 {
            amount += 0.05
            result = mix(color, with: RGB(r: 255, g: 255, b: 255), amount: amount)
        }
        return result
    }

    /// Couleur finale d'une équipe, ou l'accent si le logo n'a pas de couleur exploitable.
    public static func teamColor(rgba: [UInt8]) -> RGB {
        dominant(rgba: rgba).map(readable) ?? .accent
    }

    static func hue(_ r: Double, _ g: Double, _ b: Double, maxChannel: Double, minChannel: Double) -> Double {
        let delta = maxChannel - minChannel
        guard delta > 0 else { return 0 }
        var hue: Double
        if maxChannel == r {
            hue = (g - b) / delta
        } else if maxChannel == g {
            hue = (b - r) / delta + 2
        } else {
            hue = (r - g) / delta + 4
        }
        hue *= 60
        return hue < 0 ? hue + 360 : hue
    }

    static func mix(_ a: RGB, with b: RGB, amount: Double) -> RGB {
        func channel(_ x: UInt8, _ y: UInt8) -> UInt8 {
            UInt8((Double(x) + (Double(y) - Double(x)) * min(amount, 1)).rounded())
        }
        return RGB(r: channel(a.r, b.r), g: channel(a.g, b.g), b: channel(a.b, b.b))
    }
}

import Foundation
import LoLKit

/// `lolkit` : interroge la vraie API lolesports depuis le terminal (Windows via Docker).
@main
struct LoLKitCLI {
    static func main() async {
        let arguments = Array(CommandLine.arguments.dropFirst())
        let client = LoLClient()
        do {
            switch arguments.first {
            case "leagues":
                try await leagues(client)
            case "live":
                try await live(client)
            case "schedule" where arguments.count > 1:
                try await schedule(client, code: arguments[1])
            default:
                printUsage()
            }
        } catch {
            print("Erreur : \(error)")
            exit(1)
        }
    }

    static func printUsage() {
        print("""
        Usage :
          lolkit leagues            liste des ligues
          lolkit live               matchs en direct et leurs stats
          lolkit schedule <CODE>    dernier et prochain match d'une équipe (ex. KC)
        """)
    }

    static func leagues(_ client: LoLClient) async throws {
        for league in try await client.leagues() {
            print(league.slug.padding(toLength: 28, withPad: " ", startingAt: 0) + league.name)
        }
    }

    static func live(_ client: LoLClient) async throws {
        let events = try await client.live()
        if events.isEmpty {
            print("Aucun match en direct.")
            return
        }
        for event in events {
            let teams = event.match.teams
            print("\n\(event.league.name) · Bo\(event.bestOf) · \(teams.map(\.code).joined(separator: " vs ")) · \(teams.map { String($0.gameWins) }.joined(separator: "-"))")
            if event.statsDisabled {
                print("  Stats live non fournies par Riot pour ce match.")
                continue
            }
            guard let game = event.match.games?.first(where: { $0.state == .inProgress }) else {
                print("  Entre deux games.")
                continue
            }
            do {
                let window = try await client.window(gameId: game.id, startingTime: LiveStatsTime.startingTime(now: Date()))
                guard let snapshot = window.latestSnapshot() else {
                    print("  Pas encore de frames.")
                    continue
                }
                print("  Game \(game.number) · \(snapshot.phase.rawValue) · frame \(LoLDate.format(snapshot.timestamp))")
                for team in teams {
                    if let id = team.id, let stats = snapshot.stats(for: id) {
                        print("  " + team.code.padding(toLength: 6, withPad: " ", startingAt: 0) + describe(stats))
                    }
                }
            } catch LoLAPIError.statsDisabled {
                print("  Stats désactivées (404).")
            } catch LoLAPIError.noData {
                print("  Flux vide pour l'instant.")
            }
        }
    }

    static func schedule(_ client: LoLClient, code: String) async throws {
        let catalog = try await client.teams()
        guard let team = catalog.first(where: { $0.isActive && $0.code.lowercased() == code.lowercased() }) else {
            print("Équipe \(code) introuvable.")
            return
        }
        var follows = FollowList()
        follows.follow(team)
        let leagueIds = ScheduleQuery.leagueIds(follows: follows, leagues: try await client.leagues())
        let events = try await client.schedule(leagueIds: leagueIds)
        let card = TeamCards.build(follows: follows, schedule: events, live: try await client.live()).first
        print("\(team.name) (\(team.homeLeague?.name ?? "sans ligue d'attache"))")
        if let live = card?.live {
            print("  En direct : \(live.match.teams.map(\.code).joined(separator: " vs "))")
        }
        if let last = card?.last {
            print("  Dernier   : \(last.match.teams.map { "\($0.code) \($0.gameWins)" }.joined(separator: " - ")) · \(Format.matchDate(last.startTime))")
        }
        if let next = card?.next {
            print("  Prochain  : \(next.match.teams.map(\.code).joined(separator: " vs ")) · \(Format.matchDate(next.startTime))")
        }
        if card?.live == nil && card?.last == nil && card?.next == nil {
            print("  Aucun match dans le calendrier récent.")
        }
    }

    static func describe(_ stats: TeamStats) -> String {
        "\(stats.kills) kills · \(Format.gold(stats.gold)) gold · \(stats.towers) tours · \(stats.inhibitors) inhibs · \(stats.barons) barons · drakes [\(stats.dragons.joined(separator: ", "))]"
    }
}
